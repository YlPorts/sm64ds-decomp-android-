cmake_minimum_required(VERSION 3.22)
project(sm64ds_native_platform LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "This integration target preserves the game's 32-bit layout")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${REPO_ROOT}/android/native" native)
# Compile the actual port platform, not copies or replacement graphics routines.
# rt.cpp alone is generated by the reviewed fiber adapter from the prior stage.
file(GLOB NTR_SOURCES CONFIGURE_DEPENDS "${REPO_ROOT}/port/ntr/*.cpp")
list(REMOVE_ITEM NTR_SOURCES "${REPO_ROOT}/port/ntr/rt.cpp")
set(PORTABLE_RT "${CMAKE_CURRENT_BINARY_DIR}/rt_platform.cpp")
add_custom_command(OUTPUT "${PORTABLE_RT}"
    COMMAND "${Python3_EXECUTABLE}" "${REPO_ROOT}/android/native/tools/adapt_runtime.py"
        "${REPO_ROOT}/port/ntr/rt.cpp" "${PORTABLE_RT}"
    DEPENDS "${REPO_ROOT}/port/ntr/rt.cpp" "${REPO_ROOT}/android/native/tools/adapt_runtime.py"
    VERBATIM)
add_library(sm64ds_ntr_platform STATIC ${NTR_SOURCES} "${PORTABLE_RT}")
target_include_directories(sm64ds_ntr_platform PUBLIC "${REPO_ROOT}/port/ntr/include"
    PRIVATE "${REPO_ROOT}/port" "${REPO_ROOT}/port/hal" "${REPO_ROOT}/include")
target_compile_definitions(sm64ds_ntr_platform PRIVATE SM64DS_PLATFORM_PC=1 SM64DS_NATIVE_FIBERS=1)
target_compile_options(sm64ds_ntr_platform PRIVATE -fno-strict-aliasing -ffunction-sections -fdata-sections)
target_link_libraries(sm64ds_ntr_platform PUBLIC sm64ds_fibers Threads::Threads)
add_executable(sm64ds_platform_probe tests/platform_probe.cpp)
target_link_libraries(sm64ds_platform_probe PRIVATE sm64ds_ntr_platform)
target_compile_options(sm64ds_platform_probe PRIVATE -Wall -Wextra -Werror)
target_link_options(sm64ds_platform_probe PRIVATE -Wl,--gc-sections -Wl,--no-undefined)
add_test(NAME real_platform_pipeline COMMAND sm64ds_platform_probe)
set_tests_properties(real_platform_pipeline PROPERTIES TIMEOUT 60)
list(LENGTH NTR_SOURCES NTR_COUNT)
message(STATUS "Actual upstream NTR files: ${NTR_COUNT} plus adapted rt.cpp. NOT A COMPLETE GAME TARGET.")
