cmake_minimum_required(VERSION 3.22)
project(sm64ds_controls LANGUAGES CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "Use test_local.sh for host tests")
endif()
# This diagnostic shares the game's 32-bit pad ABI; it does not convert the engine to ARM64.
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "The original PortPadState ABI is 32 bit")
endif()
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_library(sm64ds_controls SHARED native/bridge.cpp "${ROOT}/android/full-engine/pad_android.cpp")
target_compile_features(sm64ds_controls PRIVATE cxx_std_17)
target_include_directories(sm64ds_controls PRIVATE "${ROOT}/port" "${ROOT}/android/full-engine")
target_compile_options(sm64ds_controls PRIVATE -Wall -Wextra -fno-strict-aliasing)
target_link_options(sm64ds_controls PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
target_link_libraries(sm64ds_controls PRIVATE android)
