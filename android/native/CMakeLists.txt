cmake_minimum_required(VERSION 3.22)
project(sm64ds_android_bootstrap LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
add_library(sm64ds_fibers STATIC src/fiber.cpp src/context.S)
target_include_directories(sm64ds_fibers PUBLIC include)
target_link_libraries(sm64ds_fibers PUBLIC Threads::Threads)
target_compile_options(sm64ds_fibers PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Werror;-fno-omit-frame-pointer>)
add_executable(sm64ds_fiber_tests tests/fiber_test.cpp)
target_link_libraries(sm64ds_fiber_tests PRIVATE sm64ds_fibers)
if(ANDROID AND ANDROID_ABI STREQUAL "arm64-v8a")
    target_link_options(sm64ds_fiber_tests PRIVATE "-Wl,-z,max-page-size=16384")
endif()
enable_testing()
add_test(NAME fibers COMMAND sm64ds_fiber_tests)
set_tests_properties(fibers PROPERTIES TIMEOUT 30)
# This flag NEVER enables the full game. It checks the actual upstream frame loop
# against controlled I/O/IRQ test doubles; no ROM or Nintendo asset is required.
option(SM64DS_TEST_UPSTREAM_RUNTIME "Build upstream frame-loop integration probe" OFF)
if(SM64DS_TEST_UPSTREAM_RUNTIME)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
    set(ADAPTED_RT "${CMAKE_CURRENT_BINARY_DIR}/rt_portable.cpp")
    add_custom_command(OUTPUT "${ADAPTED_RT}"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/adapt_runtime.py"
          "${REPO_ROOT}/port/ntr/rt.cpp" "${ADAPTED_RT}"
        DEPENDS "${REPO_ROOT}/port/ntr/rt.cpp" tools/adapt_runtime.py VERBATIM)
    add_executable(sm64ds_runtime_tests tests/runtime_test.cpp "${ADAPTED_RT}")
    target_include_directories(sm64ds_runtime_tests PRIVATE "${REPO_ROOT}/port/ntr/include")
    target_compile_definitions(sm64ds_runtime_tests PRIVATE SM64DS_NATIVE_FIBERS=1)
    target_link_libraries(sm64ds_runtime_tests PRIVATE sm64ds_fibers)
    add_test(NAME upstream_runtime COMMAND sm64ds_runtime_tests)
    set_tests_properties(upstream_runtime PROPERTIES TIMEOUT 30)
endif()

include(scheduler.cmake)
